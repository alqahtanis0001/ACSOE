def build(tallies=GOOD_TALLIES, refs=("shap/RUN/1/STORJ%2FUSD.parquet", "shap/RUN/2/DOGE%2FUSD.parquet"),
          files=True) -> tuple[Path, Path]:
    if WORK.exists():
        shutil.rmtree(WORK)
    out, derived = WORK / "out", WORK / "derived"
    out.mkdir(parents=True)
    for name in ("a.sqlite", "b.sqlite", "split.sqlite"):
        c = sqlite3.connect(out / name)
        c.execute("CREATE TABLE runs (run_id TEXT, scenario_digest TEXT, scenario_description TEXT)")
        c.execute("INSERT INTO runs VALUES ('RUN', 'abc', 'declared')")
        c.execute("CREATE TABLE rejections (shap_ref TEXT)")
        c.executemany("INSERT INTO rejections VALUES (?)", [(r,) for r in refs] + [(None,)])
        c.execute("CREATE TABLE scout_tallies (cycle_id, scanned, rank_feature, rank_descending, candidate, ranked, rank_run_ids)")
        c.executemany("INSERT INTO scout_tallies VALUES (?,?,?,?,?,?,?)", tallies)
        c.commit()
        c.close()
    if files:
        for r in refs:
            p = derived / r
            p.parent.mkdir(parents=True, exist_ok=True)
            p.write_bytes(b"x")
    return out, derived


def run(report: dict, **kw) -> tuple[int, str]:
    out, derived = build(**kw)
    (out / "report.json").write_text(json.dumps(report), encoding="utf-8")
    done = subprocess.run([sys.executable, str(CHECK), "rehearsal", str(out), "--derived", str(derived)],
                          capture_output=True, text=True)
    return done.returncode, done.stdout


def mutated(path: list, value) -> dict:
    r = copy.deepcopy(GOOD_REPORT)
    node = r
    for key in path[:-1]:
        node = node[key]
    if value is DELETE:
        del node[path[-1]]
    else:
        node[path[-1]] = value
    return r
